Start building in minutes

Connect with Redis using trusted libraries in your language of choice.

import { createClient } from 'redis';

const client = createClient();
await client.connect();

await client.json.set('user:1', '$', {
  name: 'Alice',
  emails: ['alice@example.com', 'alice@work.com'],
  address: { city: 'NYC', zip: '10001' }
});

console.log(await client.json.get('user:1', { path: '$' }));            // { name: 'Alice', emails: [...], address: {...} }
console.log(await client.json.get('user:1', { path: '$.name' }));       // "Alice"
console.log(await client.json.get('user:1', { path: '$.emails[0]' }));  // "alice@example.com"
console.log(await client.json.get('user:1', { path: '$.address.zip' })); // "10001"

await client.quit();

import redis.clients.jedis.Jedis;
import redis.clients.jedis.JedisPubSub;

// --- Subscriber Class ---
class Subscriber extends JedisPubSub {
    @Override
    public void onMessage(String channel, String message) {
        System.out.println("Received message: " + message); // Received message: Hello, World!
        this.unsubscribe();
    }
}

// --- Main Class (Publisher + Subscriber Starter) ---
public class PubSubExample {
    public static void main(String[] args) {
        Jedis subscriberJedis = new Jedis("localhost", 6379);
        Jedis publisherJedis = new Jedis("localhost", 6379);

        Subscriber subscriber = new Subscriber();

        new Thread(() -> subscriberJedis.subscribe(subscriber, "my-channel")).start();

        try {
            Thread.sleep(100); // Give the subscriber a moment to connect
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        publisherJedis.publish("my-channel", "Hello, World!"); // Publish message

        publisherJedis.close();
        subscriberJedis.close();
    }
}

<?php
require 'vendor/autoload.php';

$redis = new Predis\Client();
$mode = $argv[1] ?? null;

if ($mode === 'produce') {
    $job = ['task' => 'email', 'to' => 'user@example.com'];
    $redis->lpush('queue', json_encode($job));
    echo "Job added.\n";
} elseif ($mode === 'consume') {
    echo "Waiting for jobs...\n";
    while (true) {
        $job = $redis->brpop('queue', 0);
        $data = json_decode($job[1], true);
        echo "Processing {$data['task']} to {$data['to']}\n";
    }
} else {
    echo "Usage: php queue.php [produce|consume]\n";
}

package main

import (
	"context"
	"fmt"
	"github.com/redis/go-redis/v9"
)

var ctx = context.Background()

func main() {
	rdb := redis.NewClient(&redis.Options{
		Addr: "localhost:6379",
	})

	// Increment the counter
	val, err := rdb.Incr(ctx, "pageviews").Result()
	if err != nil {
		panic(err)
	}

	fmt.Printf("Pageviews: %d\n", val) // Pageviews: 1 (then 2, 3, ...)
}
